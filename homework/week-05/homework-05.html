def classify5(train, labels, test, distance='euclidean'):
    predictions = []
    for image in test:
        difference = train-image
        if distance == 'euclidean':
            d = np.sqrt(np.sum(difference**2, axis=1))
        elif distance == 'manhattan':
            d = np.sum(np.abs(difference), axis=1)
        else:
            raise ValueError('Unknown distance')
        neighbors = np.argsort(d, kind='stable')[:5]
        votes = np.bincount(labels[neighbors], minlength=10)
        predictions.append(np.argmax(votes))
    return np.array(predictions)

import numpy as np
from scipy.spatial.distance import cdist
from sklearn.linear_model import Lasso

def f(x):
    return 0.5*x[:, 0] + np.sin(x[:, 1]) - 0.3*x[:, 2]**2


def regress_all_k(train, y, test, ks):
    order = np.argsort(cdist(test, train, metric='sqeuclidean'), axis=1, kind='stable')
    sums = np.cumsum(y[order], axis=1)
    return sums[:, ks-1] / ks


# Q1: one data set, followed by 200 fresh data sets.
rng = np.random.default_rng(5051)
ks1 = np.arange(1, 30, 2)
x0 = np.array([[0.5, 0.7, 1.]])
truth = f(x0)[0]


def prediction_q1():
    x = rng.normal(size=(400, 3))
    y = f(x) + rng.normal(0, 0.1, 400)
    return regress_all_k(x, y, x0, ks1)[0]

# Q2: manual fold-specific scaling and response centering.
rng = np.random.default_rng(5052)
penalties = np.logspace(-4, 0, 41)
rows2 = []
for rho in [0., 0.8]:
    covariance = rho*np.ones((30, 30)) + (1-rho)*np.eye(30)
    x = rng.normal(size=(1400, 30)) @ np.linalg.cholesky(covariance).T
    y = f(x) + rng.normal(0, 0.1, 1400)
    train, test = x[:400], x[400:]
    yt, ytest = y[:400], y[400:]
    knn_pred = regress_all_k(train, yt, test, np.array([5]))[:, 0]
    fold = rng.permutation(np.repeat(np.arange(10), 40))
    errors = np.empty((10, 41))
    for j in range(10):
        fit, valid = fold != j, fold == j
        center = train[fit].mean(axis=0)
        scale = train[fit].std(axis=0, ddof=0)
        xf = (train[fit]-center)/scale
        xv = (train[valid]-center)/scale
        ymean = yt[fit].mean()
        for h, penalty in enumerate(penalties):
            model = Lasso(alpha=penalty, fit_intercept=False, tol=1e-8, max_iter=100000)
            model.fit(xf, yt[fit]-ymean)
            errors[j, h] = np.mean((yt[valid] - model.predict(xv)-ymean)**2)
    penalty = penalties[np.argmin(errors.mean(axis=0))]
    center, scale, ymean = train.mean(axis=0), train.std(axis=0, ddof=0), yt.mean()
    model = Lasso(alpha=penalty, fit_intercept=False, tol=1e-8, max_iter=100000)
    model.fit((train-center)/scale, yt-ymean)
    lasso_pred = model.predict((test-center)/scale) + ymean
    rows2.append(['Independent' if rho == 0 else 'Correlated', np.mean((knn_pred-ytest)**2), np.mean((lasso_pred-ytest)**2), penalty, np.count_nonzero(model.coef_)])

print(rows2)
